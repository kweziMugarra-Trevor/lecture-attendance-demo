@app.route('/join/<sid>', methods=['GET','POST'])
def join(sid):
    s = get_session(sid)
    error = None
    success = False
    if request.method == 'POST':
        # Basic per-IP throttling; not a guarantee against distributed guessing.
        now = time.time()
        ip = request.remote_addr or 'unknown'
        queue = attempts[ip]
        while queue and queue[0] < now - 60:
            queue.popleft()
        if len(queue) >= 12:
            return render_template('join.html', s=s, error='Too many attempts. Wait one minute and try again.', success=False), 429
        queue.append(now)
        try:
            record(sid, request.form.get('student_name'), request.form.get('student_number'), 'phone', request.form.get('pin'))
            success = True
        except ValueError as exc:
            error = str(exc)
    return render_template('join.html', s=s, error=error, success=success)
